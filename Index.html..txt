<script>
let device=null;
let server=null;

async function connectScooter(){
try{
if(!navigator.bluetooth){
throw new Error("Web Bluetooth nicht verfügbar");
}

document.getElementById("connectionInfo").textContent="Suche...";
device=await navigator.bluetooth.requestDevice({
acceptAllDevices:true,
optionalServices:["battery_service"]
});

document.getElementById("deviceName").textContent=device.name||"NAVEE";
server=await device.gatt.connect();

document.getElementById("status").textContent="● CONNECTED";
document.getElementById("status").style.color="#36e68b";
document.getElementById("connectionInfo").textContent="Bluetooth verbunden";

device.addEventListener("gattserverdisconnected",()=>{
document.getElementById("status").textContent="● OFFLINE";
document.getElementById("connectionInfo").textContent="Getrennt";
});

await discoverServices();

}catch(e){
document.getElementById("connectionInfo").textContent=e.message;
}
}

async function discoverServices(){
try{
let output="";
let services=await server.getPrimaryServices();

for(const service of services){
output+="<br>Service: "+service.uuid;
let chars=await service.getCharacteristics();

for(const c of chars){
output+="<br>Characteristic: "+c.uuid;
output+=" ["+Object.keys(c.properties).filter(k=>c.properties[k]).join(", ")+"]";

if(c.properties.read){
try{
let data=await c.readValue();
let bytes=Array.from(new Uint8Array(data.buffer,data.byteOffset,data.byteLength));
output+="<br>DATA: "+bytes.map(x=>x.toString(16).padStart(2,"0")).join(" ");
}catch(e){}
}
}
}
document.getElementById("debug").innerHTML=output||"Keine Services gefunden";
}catch(e){
document.getElementById("debug").textContent=e.message;
}
}

function disconnectScooter(){
if(device&&device.gatt.connected)device.gatt.disconnect();
}
